// === Google Apps Script สำหรับรับเรื่องร้องเรียน === // วางโค้ดนี้ใน Apps Script ของ Google Sheet function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); sheet.appendRow([ data.timestamp, data.fullname, data.phone, data.email, data.contact, data.area, data.address, data.mapLink, data.category, data.details, data.urgency, data.previousReport, data.previousReportDetails, data.suggestions, data.images ]); // ส่งอีเมลแจ้งเตือน (เปลี่ยนอีเมลตามต้องการ) var emailTo = 'am.kaweekate@gmail.com'; // ⚠️ เปลี่ยนเป็นอีเมลของคุณ var subject = '[ร้องเรียน-เขต3] ' + data.category + ' - ' + data.area + ' (' + data.urgency + ')'; var body = 'มีเรื่องร้องเรียนใหม่เข้ามา\n\n' + 'ชื่อ: ' + data.fullname + '\n' + 'เบอร์โทร: ' + data.phone + '\n' + 'อีเมล: ' + data.email + '\n' + 'พื้นที่: ' + data.area + '\n' + 'ประเภท: ' + data.category + '\n' + 'ความเร่งด่วน: ' + data.urgency + '\n\n' + 'รายละเอียด:\n' + data.details + '\n\n' + '📊 ดูข้อมูลทั้งหมดได้ที่ Google Sheet'; MailApp.sendEmail(emailTo, subject, body); return ContentService .createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ทดสอบ: รันฟังก์ชันนี้เพื่อเช็คว่า Sheet เชื่อมต่อได้ function testSetup() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); Logger.log('Sheet name: ' + sheet.getName()); Logger.log('ระบบพร้อมใช้งาน ✅'); }
